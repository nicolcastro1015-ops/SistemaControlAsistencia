-- =====================================================================
-- 2. TABLA USUARIO
--    Relación: USUARIO 1 : N ASISTENCIA
-- =====================================================================
CREATE TABLE dbo.Usuario
(
    IdUsuario        INT IDENTITY(1,1) NOT NULL,
    Nombre           NVARCHAR(100)     NOT NULL,
    Apellidos        NVARCHAR(150)     NOT NULL,
    Correo           NVARCHAR(150)     NOT NULL,
    ContrasenaHash   NVARCHAR(300)     NOT NULL,
    Rol              NVARCHAR(20)      NOT NULL,
    Estado           BIT               NOT NULL CONSTRAINT DF_Usuario_Estado DEFAULT (1),

    CONSTRAINT PK_Usuario PRIMARY KEY (IdUsuario),
    CONSTRAINT UQ_Usuario_Correo UNIQUE (Correo),
    CONSTRAINT CK_Usuario_Rol CHECK (Rol IN ('Administrador', 'Empleado'))
);
GO

-- =====================================================================
-- 3. TABLA ASISTENCIA
--    Cada registro pertenece a un único usuario (FOREIGN KEY).
--    No se permiten asistencias de usuarios inexistentes.
--    ON DELETE CASCADE: al eliminar un usuario (GU-03), SQL Server elimina
--    automáticamente todos sus registros de asistencia asociados, para que
--    la eliminación desde la aplicación (DELETE FROM Usuario) nunca sea
--    rechazada por dejar registros huérfanos en esta tabla.
-- =====================================================================
CREATE TABLE dbo.Asistencia
(
    IdAsistencia   INT IDENTITY(1,1) NOT NULL,
    IdUsuario      INT               NOT NULL,
    TipoRegistro   NVARCHAR(10)      NOT NULL,
    FechaHora      DATETIME2(0)      NOT NULL,

    CONSTRAINT PK_Asistencia PRIMARY KEY (IdAsistencia),
    CONSTRAINT FK_Asistencia_Usuario FOREIGN KEY (IdUsuario)
        REFERENCES dbo.Usuario (IdUsuario)
        ON DELETE CASCADE,
    CONSTRAINT CK_Asistencia_TipoRegistro CHECK (TipoRegistro IN ('Entrada', 'Salida'))
);
GO

-- Índice de apoyo: los reportes y validaciones siempre filtran por usuario + fecha.
CREATE INDEX IX_Asistencia_IdUsuario_FechaHora ON dbo.Asistencia (IdUsuario, FechaHora);
GO

-- =====================================================================
-- 7. CONSULTAS SQL DE COMPROBACIÓN
--    Sirven para demostrar al docente que el modelo relacional permite
--    resolver cada requerimiento. Usan las mismas tablas y nombres que
--    utiliza posteriormente el código C# (Data/UsuarioRepository.cs y
--    Data/AsistenciaRepository.cs).
-- =====================================================================

-- 7.1 Todos los usuarios
SELECT IdUsuario, Nombre, Apellidos, Correo, Rol, Estado FROM dbo.Usuario ORDER BY Apellidos;

-- 7.2 Usuarios activos
SELECT IdUsuario, Nombre, Apellidos, Correo, Rol FROM dbo.Usuario WHERE Estado = 1;

-- 7.3 Buscar usuario por correo
SELECT * FROM dbo.Usuario WHERE Correo = 'ana.torres@empresa.cl';

-- 7.4 Todos los registros de asistencia
SELECT a.IdAsistencia, u.Nombre, u.Apellidos, a.TipoRegistro, a.FechaHora
FROM dbo.Asistencia a
INNER JOIN dbo.Usuario u ON u.IdUsuario = a.IdUsuario
ORDER BY a.FechaHora;

-- 7.5 Asistencias de una fecha determinada
SELECT u.Nombre, u.Apellidos, a.TipoRegistro, a.FechaHora
FROM dbo.Asistencia a
INNER JOIN dbo.Usuario u ON u.IdUsuario = a.IdUsuario
WHERE CAST(a.FechaHora AS DATE) = '2026-08-24'
ORDER BY a.FechaHora;

-- 7.6 Asistencias de un trabajador específico
SELECT a.TipoRegistro, a.FechaHora
FROM dbo.Asistencia a
INNER JOIN dbo.Usuario u ON u.IdUsuario = a.IdUsuario
WHERE u.Correo = 'ana.torres@empresa.cl'
ORDER BY a.FechaHora DESC;

-- 7.7 RE-01: Entradas posteriores a las 09:30 (atrasos). 09:30 exacto NO cuenta.
SELECT u.IdUsuario, u.Nombre, u.Apellidos, CAST(a.FechaHora AS DATE) AS Fecha, CAST(a.FechaHora AS TIME) AS HoraEntrada
FROM dbo.Asistencia a
INNER JOIN dbo.Usuario u ON u.IdUsuario = a.IdUsuario
WHERE a.TipoRegistro = 'Entrada' AND CAST(a.FechaHora AS TIME) > '09:30:00'
ORDER BY u.Apellidos;

-- 7.8 RE-02: Salidas anteriores a las 17:30 (salidas anticipadas). 17:30 exacto NO cuenta.
SELECT u.IdUsuario, u.Nombre, u.Apellidos, CAST(a.FechaHora AS DATE) AS Fecha, CAST(a.FechaHora AS TIME) AS HoraSalida
FROM dbo.Asistencia a
INNER JOIN dbo.Usuario u ON u.IdUsuario = a.IdUsuario
WHERE a.TipoRegistro = 'Salida' AND CAST(a.FechaHora AS TIME) < '17:30:00'
ORDER BY u.Apellidos;

-- 7.9 RE-03: Inasistencias de una fecha determinada
--     (usuarios activos que NO tienen ningún registro ese día)
SELECT u.IdUsuario, u.Nombre, u.Apellidos, '2026-08-24' AS Fecha
FROM dbo.Usuario u
WHERE u.Estado = 1
  AND NOT EXISTS (
        SELECT 1 FROM dbo.Asistencia a
        WHERE a.IdUsuario = u.IdUsuario
          AND CAST(a.FechaHora AS DATE) = '2026-08-24'
  )
ORDER BY u.Apellidos;

-- 7.10 Total de empleados activos
SELECT COUNT(*) AS TotalEmpleadosActivos FROM dbo.Usuario WHERE Estado = 1;

-- 7.11 Presentes del día (usuarios distintos con al menos una Entrada en la fecha)
SELECT COUNT(DISTINCT a.IdUsuario) AS PresentesHoy
FROM dbo.Asistencia a
WHERE a.TipoRegistro = 'Entrada' AND CAST(a.FechaHora AS DATE) = '2026-08-24';

-- 7.12 Atrasos del día (reutiliza la misma condición de 7.7 para la fecha de hoy)
SELECT COUNT(*) AS AtrasosHoy
FROM dbo.Asistencia a
WHERE a.TipoRegistro = 'Entrada'
  AND CAST(a.FechaHora AS DATE) = '2026-08-24'
  AND CAST(a.FechaHora AS TIME) > '09:30:00';

-- 7.13 Salidas anticipadas del día
SELECT COUNT(*) AS SalidasAnticipadasHoy
FROM dbo.Asistencia a
WHERE a.TipoRegistro = 'Salida'
  AND CAST(a.FechaHora AS DATE) = '2026-08-24'
  AND CAST(a.FechaHora AS TIME) < '17:30:00';

-- 7.14 Actividad reciente (últimos 10 movimientos registrados)
SELECT TOP (10) u.Nombre, u.Apellidos, a.TipoRegistro, a.FechaHora
FROM dbo.Asistencia a
INNER JOIN dbo.Usuario u ON u.IdUsuario = a.IdUsuario
ORDER BY a.FechaHora DESC;
GO
